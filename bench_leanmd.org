*** Preprocessing step

#+name: preprocess
#+BEGIN_SRC yaml :eval no
  - name: preprocess
    tag: base
    _: |
      $modules
      export LD_LIBRARY_PATH=/work/casch/manual_install/install/lib:$LD_LIBRARY_PATH
      export PKG_CONFIG_PATH=/work/casch/manual_install/install/lib/pkgconfig:$PKG_CONFIG_PATH
      export PATH=/work/casch/manual_install/install/bin:$PATH
      sed 4d Makefile -i
      sed 4i"CHARMBASE=/work/casch/manual_install/install" Makefile -i
      make -j
  - name: preprocess
    tag: tracing
    _: |
      $modules
      export LD_LIBRARY_PATH=/work/casch/manual_install/install/lib:$LD_LIBRARY_PATH
      export PKG_CONFIG_PATH=/work/casch/manual_install/install/lib/pkgconfig:$PKG_CONFIG_PATH

* Steps

** Running the program

#+name: run_step
#+BEGIN_SRC yaml :eval no
  - name: run-program
    use:
      - globalParameter
      - systemParameter
      - inputset
      - executeset
      - source_dir
      - from: platform.xml
        _: jobfiles
      - from: platform.xml
        _: executesub
    do:
      done_file: $done_file
      _: $submit $submit_script
#+END_SRC
